// one-off evidence for the report (not committed): each index's account order for its headline fund, without and with D64
import { it } from "vitest";
import { writeFileSync } from "node:fs";
import { buildInvest, knownIndices, indexById } from "./invest";
import { buildWorkbenchBundle } from "./providers.server";
import type { WorkbenchInputs } from "./providers";
import { entered } from "./value";

const bundle = buildWorkbenchBundle();
const t = bundle.today;
const plan = (amountNow: number, monthly: number, over: Partial<WorkbenchInputs> = {}): WorkbenchInputs => ({
  amountNow, monthly, years: 20, grossReturn: 0.07, inflation: 0.02, age: null, income: null,
  values: { amount: entered(amountNow, "nis", t), monthly: entered(monthly, "nis_per_month", t), years: entered(20, "years", t), returnV: entered(7, "pct_per_year", t), inflation: entered(2, "pct_per_year", t) },
  fromGoal: false, returnFromParameter: false, ...over,
});

it("evidence", () => {
  const out: unknown[] = [];
  for (const [amount, monthly] of [[6000, 400], [100000, 0]] as const) {
    for (const k of knownIndices(bundle)) {
      const index = indexById(bundle, k.id)!;
      const runs = [["before", plan(amount, monthly)], ["after", plan(amount, monthly, { currentAccounts: { own: null } })], ["after, own leumi", plan(amount, monthly, { currentAccounts: { own: "leumi" } })]] as const;
      const row: Record<string, unknown> = { index: k.id, plan: `${amount}+${monthly}/mo, 20y, 7%` };
      for (const [name, p] of runs) {
        const v = buildInvest({ bundle, index, plan: p });
        row[name] = {
          headline: v.headline ? `${v.headline.providerId} ${v.headline.fund.rec.id} ${Math.round(v.headline.valueN)}` : null,
          houses: v.houses.filter((h) => h.best).map((h) => `${h.provider.id}:${Math.round(h.best!.price!.valueN)}`),
        };
      }
      out.push(row);
    }
  }
  writeFileSync(process.env.OUT!, JSON.stringify(out, null, 1));
});
